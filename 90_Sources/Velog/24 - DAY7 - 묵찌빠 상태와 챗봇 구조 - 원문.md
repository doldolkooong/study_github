**1. 묵찌빠 게임 구현**

<코드 어떤 식으로 진행할 건지>
>처음에는 가위 바위 보로 시작한다.
가위 바위 보 결과에 따라 공격 결정
	- user 승리 -> user 공격
	- computer 승리 -> computer 공격
공격권을 가진 사람이 가위 바위 보 진행
공격권을 가진 사람이 이기면 공격권 유지
공격권을 가진 사람이 지면 공격권을 상대에게 넘김
처음 이후 가위바위보 결과가 무승부이면 공격권 가진 사람 승리
승자가 결정되면 게임 종료

<br>

강사님께 코드 주석 관련하여 피드백을 받았다.
<br>

>1. **주어를 통일하기**
	사용자는 가위바위보를 입력한다.
	사용자는 입력값이 올바른지 확인한다.
	사용자는 입력값이 잘못된 경우 다시 입력한다.

>2. **코드를 작성하는 데 도움이 되도록 작성하기**
	사용자는 입력한 점수를 변수에 저장한다.
	저장된 점수를 이용하여 총점을 계산한다.
	총점은 sum()을 사용하여 계산한다.

<br>

**작성한 코드**
```python
import random
import enum

class rsp_choice(enum.Enum): # 가위, 바위, 보에 값을 넣은 상수 생성
    가위 = 0
    바위 = 1
    보 = 2

def mukchippa():
    attacker = None

    # 1. 누가 공격할 건지 정하기 - 가위바위보 반복문
    while True:
        computer_choice = random.choice(list(rsp_choice.__members__)) # rsp_choice에서 computer가 랜덤으로 선택
        user_choice = input("가위, 바위, 보 중 하나 적어주세요: ") # 사용자의 입력 받음

        if user_choice not in rsp_choice.__members__: # 사용자가 rsp_choice에서 선택하지 않은 경우 출력
                print("잘못 입력했습니다. 다시 입력해주세요.")
                continue
        
        print(f"User: {user_choice}") # 사용자의 선택 출력
        print(f"Computer: {computer_choice}") # 컴퓨터의 선택 출력

        # 같은 것을 냈으면 게임 진행이 안됨
        # 다시 고를 수 있도록
        if user_choice == computer_choice:
            print("====비겼습니다. 다시 가위바위보!====")
            continue

        # 가위 바위 보 계산
        result = (rsp_choice[user_choice].value - rsp_choice[computer_choice].value) % 3

        # User 승리 -> User의 공격
        if result == 1: 
            attacker = "User"
            print("User가 공격권을 가져갑니다.")
            break

        # Computer 승리 → Computer의 공격
        elif result == 2:
            attacker = "Computer"
            print("====Computer가 공격권을 가져갑니다.====")
            break


    # 2. 묵찌빠 진행 반복문
    while True:
        print(f"<현재 공격권 : {attacker}>") # 공격 할 사람 저장

        user_choice = input("가위, 바위, 보 중 하나 적어주세요: ") # 사용자의 입력 받음
        computer_choice = random.choice(list(rsp_choice.__members__))

        if user_choice not in rsp_choice.__members__: # user가 rsp_choice에 없는 것을 입력할 경우
            print("잘못 입력했습니다. 다시 입력해주세요.")
            continue

        print(f"User: {user_choice}")
        print(f"Computer: {computer_choice}")

        result = (rsp_choice[user_choice].value - rsp_choice[computer_choice].value) % 3

        if result == 1: # user의 승리 -> 공격권은 user에게
            attacker = "User" # 공격권에 User 저장
            print("====User가 공격권을 가져갑니다.====")

        elif result == 2:  # computer의 승리 -> 공격권은 computer에게
            attacker = "Computer" # 공격권에 Computer 저장
            print("====Computer가 공격권을 가져갑니다.====")

        else: # 만약 같은 것을 내면 attacker의 승리
            print(f"~~{attacker}의 승리~~")
            break


mukchippa()
```

``` python
import streamlit as st

from .constants import ROLE

def show_msg(role:ROLE, msg:str, is_history = False) -> None:

    # 검증로직
    assert isinstance(role, ROLE), "정상적인 코드가 아닙니다."

    with st.chat_message(role.name): # 화면에 뿌려지기만 하면 됨 -> 함수
        st.markdown(msg)

    # app에서는 append 되어야 하지만
    # history에서는 append 되어서는 안됨
    if not is_history:
        st.session_state.history.append({
            "role" : role, # 객체
            "msg" : msg # 문자열 

        })
  ```

4) history.py (이전에 대화한 내용 저장하는 로직)

``` python
import streamlit as st
from common.show import show_msg # 함수

# session_state는 딕셔너리 상태
# 딕셔너리는 키가 유니크 해야됨, 세트는 중복이 안됨, 튜플은 수정이 안됨
# 이력 데이터를 저장하기 위해서 선언
# -> 초기화랑 같은 말임

def init_history() -> None:
    # 이미 저장하고 있기 때문에 클래스가 아닌 함수
    if "history" not in st.session_state: # 내용이 st.session_state에 없으면 리스트 초기화
        st.session_state.history = [] 

    for h in st.session_state.history:
        show_msg(**h, is_history=True)
```

 ``` python
from groq import Groq
from dotenv import load_dotenv
import streamlit as st
from .constants import ROLE

# clinet는 네트워크 인터넷 통신할 때 사용하는 용어
@st.cache_resource # Singleton 디자인 패턴 적용
def get_client():
    load_dotenv() # .env 파일에 등록된 데이터 환경변수에 저장하는 함수
    return Groq()
#                    input형태   기존 코드는 input만, but 가변 파라미터 해서 상관 없음   output형태
def get_msg_of_ai(
    user_input:str, model_name:str="openai/gpt-oss-120b") -> str:

    # 이력 데이터 추가
    messages = [
        {
            "role": history['role'].name, # 객체가 아닌 문자열을 전달해야됨
            "content": history['msg']
        }for history in st.session_state.history
    ]

    # 사용자 메세지 추가
    # 기존 데이터와 현재 데이터 전부 추가
    messages.append({
            "role": ROLE.user.name,
            "content": user_input
    })

    client = get_client()
    response = client.chat.completions.create(
        messages = messages,
        model = model_name
    )

    return response.choices[0].message.content
 ```
